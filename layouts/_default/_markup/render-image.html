{{/* Copy of themes/hextra/layouts/_default/_markup/render-image.html with two additions: 1. An image that lives under assets/ is run through Hugo's image pipeline instead of being linked verbatim. The migrated Statusengine 3 screenshots are up to 3104px wide and ~800 KB each, which is several times what the ~672px content column can show. They are kept at full size in assets/images/ so the originals stay untouched, and downscaled here. 2. Processed images are wrapped in a link to a larger rendition, which se-lightbox.js turns into an overlay. The link is the point: without JavaScript it still opens the big image, so the feature degrades to something useful rather than to nothing. Images still under static/ (the logo, the two mascots) fall through to the theme's original behaviour unchanged, and are not made clickable. Keep in sync when bumping the theme. */}} {{- $alt := .PlainText | safeHTML -}} {{- $lazyLoading := .Page.Site.Params.enableImageLazyLoading | default true -}} {{- $dest := .Destination -}} {{- $isLocal := not (urls.Parse $dest).Scheme -}} {{- $isPage := and (eq .Page.Kind "page") (not .Page.BundleType) -}} {{- $startsWithSlash := hasPrefix $dest "/" -}} {{- $startsWithRelative := hasPrefix $dest "../" -}} {{/* Widest the content column ever needs, at 2x for high-density screens. */}} {{- $maxWidth := 1400 -}} {{/* What the lightbox shows. Beyond this the file grows faster than the detail a reader can actually use on screen. */}} {{- $lightboxWidth := 2400 -}} {{- $width := "" -}} {{- $height := "" -}} {{- $full := "" -}} {{- $fullWidth := "" -}} {{- $processed := false -}} {{- if and $dest $isLocal -}} {{- with resources.Get (strings.TrimPrefix "/" $dest) -}} {{- if ne .MediaType.SubType "svg" -}} {{- $native := .Width -}} {{/* Resizing to its own width is a no-op geometrically but still re-encodes to WebP, so every branch ends up in the same format. */}} {{- $target := cond (gt $native $maxWidth) $maxWidth $native -}} {{- with .Resize (printf "%dx webp q82" $target) -}} {{- $dest = .RelPermalink -}} {{- $width = .Width -}} {{- $height = .Height -}} {{- $processed = true -}} {{- end -}} {{- $big := cond (gt $native $lightboxWidth) $lightboxWidth $native -}} {{- with .Resize (printf "%dx webp q85" $big) -}} {{- $full = .RelPermalink -}} {{- $fullWidth = .Width -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $processed -}} {{- if $startsWithSlash -}} {{/* Images under static directory */}} {{- $dest = (relURL (strings.TrimPrefix "/" $dest)) -}} {{- else if and $isPage (not $startsWithRelative) -}} {{/* Images that are sibling to the individual page file */}} {{ $dest = (printf "../%s" $dest) }} {{- end -}} {{- end -}} {{- end -}} {{- define "partials/se-img" -}} {{- end -}} {{- $img := dict "dest" $dest "alt" $alt "title" .Title "width" $width "height" $height "lazy" $lazyLoading -}} {{- with .Title -}} {{- else -}} {{- if $full -}} {{ partial "se-img" $img }} {{- else -}} {{- partial "se-img" $img -}} {{- end -}} {{- end -}}
